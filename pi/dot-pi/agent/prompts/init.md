# /init

Self-bootstrapping project initialization. Detects and installs external dependencies, indexes the codebase for agent memory, seeds a project-specific AGENTS.md, and wires git hooks for auto-refresh.

Run once per project. Idempotent — safe to re-run to refresh.

```bash
!python3 - <<'PY'
import os
import shutil
import subprocess
from pathlib import Path


def run(command, *, cwd, capture=False, check=True, env=None):
    return subprocess.run(
        command,
        cwd=str(cwd),
        text=True,
        capture_output=capture,
        check=check,
        env=env,
    )


def fail(message):
    print(f"ERROR: {message}")
    raise SystemExit(1)


def git_root():
    result = run(["git", "rev-parse", "--show-toplevel"], cwd=Path.cwd(), capture=True, check=False)
    if result.returncode != 0:
        fail("Not a git repository. Run /init from a project directory.")
    return Path(result.stdout.strip())


def ensure_cli(name):
    if shutil.which(name):
        print(f"  ✓ {name}: already on PATH ({shutil.which(name)})")
        return True
    print(f"  ⚠ {name}: not found. Install manually and re-run /init.")
    return False


def index_codebase(root):
    if not shutil.which("agent-memory"):
        print("  ⚠ agent-memory not on PATH, skipping codebase indexing.")
        return
    print("  Indexing codebase with agent-memory ...")
    result = run(["agent-memory", "index"], cwd=root, capture=True, check=False)
    if result.returncode != 0:
        print(f"  ⚠ agent-memory index failed (non-fatal):\n{result.stdout}\n{result.stderr}")
        return
    print("  ✓ agent-memory: indexed")


def install_git_hooks(root):
    hooks_dir = root / ".git" / "hooks"
    if not hooks_dir.exists():
        print("  ⚠ .git/hooks not found, skipping git hooks.")
        return

    # Install a post-checkout hook that re-indexes agent-memory
    hook_path = hooks_dir / "post-checkout"
    hook_content = """#!/bin/bash
# Auto-generated by /init — re-index agent-memory on branch switch
if command -v agent-memory &>/dev/null; then
    agent-memory index --quiet 2>/dev/null
fi
"""
    if hook_path.exists():
        # Check if our marker is already there
        existing = hook_path.read_text(encoding="utf-8", errors="ignore")
        if "Auto-generated by /init" in existing:
            print("  ✓ git hook (post-checkout): already installed")
            return
        print("  ⚠ post-checkout hook already exists, skipping (manual merge needed)")
        return

    hook_path.write_text(hook_content, encoding="utf-8")
    hook_path.chmod(0o755)
    print("  ✓ git hook (post-checkout): installed")


def detect_stack(root):
    detected = []
    if (root / "package.json").exists():
        detected.append("Node.js / TypeScript")
    if (root / "pyproject.toml").exists() or (root / "requirements.txt").exists():
        detected.append("Python")
    if (root / "go.mod").exists():
        detected.append("Go")
    if (root / "Cargo.toml").exists():
        detected.append("Rust")
    if (root / "pom.xml").exists():
        detected.append("Java")
    if not detected:
        detected.append("Unknown")
    return sorted(detected)


def top_level_dirs(root):
    ignore = {".git", ".venv", "node_modules", "dist", "build", "target", ".context"}
    dirs = []
    for item in sorted(root.iterdir(), key=lambda item: item.name.lower()):
        if item.is_dir() and not item.name.startswith(".") and item.name not in ignore:
            dirs.append(item.name)
    return dirs


def generate_agents_md(root):
    stack_lines = "\n- ".join(detect_stack(root))
    dir_lines = "\n- ".join(top_level_dirs(root)) or "(project scaffold)"
    return f"""# AGENTS.md

Auto-generated by `/init`. Edit freely to add project-specific instructions.

## Project Structure

- Root: `{root.name}`
- Directories:
  - {dir_lines}

## Tech Stack

- {stack_lines}

## Conventions

- Keep changes scoped and small.
- Respect existing naming and file layout conventions.
- Prefer deterministic, reviewable edits over broad refactors.
- TDD: write failing test → watch it fail → make it pass → refactor.

## Architecture Decisions

- Document key tradeoffs and component boundaries here as the project evolves.
"""


def ensure_agents_md(root):
    agents_file = root / "AGENTS.md"
    if agents_file.exists():
        print(f"  ✓ AGENTS.md: already present ({agents_file})")
        return
    agents_file.write_text(generate_agents_md(root), encoding="utf-8")
    print("  ✓ AGENTS.md: written")


def ensure_context_dir(root):
    context_dir = root / ".context"
    plans_dir = context_dir / "plans"
    plans_dir.mkdir(parents=True, exist_ok=True)
    # Add .context to .gitignore if not already there
    gitignore = root / ".gitignore"
    if gitignore.exists():
        content = gitignore.read_text(encoding="utf-8")
        if ".context" not in content:
            gitignore.write_text(content.rstrip() + "\n.context/\n", encoding="utf-8")
            print("  ✓ .gitignore: added .context/")
        else:
            print("  ✓ .gitignore: .context/ already ignored")
    else:
        gitignore.write_text(".context/\n", encoding="utf-8")
        print("  ✓ .gitignore: created with .context/")


def main():
    root = git_root()

    print(f"  Project root: {root}\n")

    print("  Checking dependencies...")
    ensure_cli("agent-memory")
    ensure_cli("git")

    print("\n  Setting up project...")
    index_codebase(root)
    install_git_hooks(root)
    ensure_agents_md(root)
    ensure_context_dir(root)

    print("\n  ✅ Init complete.")
    print("\n  Next steps:")
    print("    /plan <topic>   — interactive plan builder")
    print("    /task <desc>    — plan → build → review pipeline")
    print("    /worktree       — create isolated worktree")


if __name__ == "__main__":
    main()
PY
```
